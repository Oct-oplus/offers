function act(el){

 if(a==='open')openSheet(id,el.dataset.hl);
 else if(a==='goto'){closeSheet();var p=id.split(':');S.tab=p[0];if(p[1])S.sub=p[1];render();scrollTo(0,0)}
 else if(a==='cat'){S.cat=id;S.tab='offers';render();scrollTo(0,0)}
 else if(a==='sub'){S.sub=id;render()}
 else if(a==='ack'){NOTES[+id].ack=true;toast('تم تأكيد قراءة الملاحظة');render()}
 else if(a==='remind'){if(!S.tg){toast('اربط تيليجرام أولاً لتصلك التذكيرات');S.tab='help';render();return}S.remind=!S.remind;toast(S.remind?'تم تفعيل التذكير':'تم إيقاف التذكير');render()}
 else if(a==='tg'){S.tg=!S.tg;if(!S.tg)S.remind=false;toast(S.tg?'تم ربط حسابك بالبوت (تجريبي)':'تم إلغاء الربط');render()}
 else if(a==='subscribe'){closeSheet();S.tab='help';S.pick=0;render();scrollTo(0,0);toast('اختر منسقة بلدك وأرسل لها اسم العرض')}
 else if(a==='star'){S.rate=+id;[].forEach.call(document.querySelectorAll('#stars button'),function(b,i){b.classList.toggle('on',i<S.rate)})}
 else if(a==='pub'){var t=$('#rtxt').value.trim();if(!S.rate){toast('اختر عدد النجوم أولاً');return}(REV[id]=REV[id]||[]).unshift({n:'أنت',s:S.rate,t:t||'—'});toast('تم نشر تقييمك');openSheet(id,1);render()}
 else if(a==='fb'){if(!$('#ftxt').value.trim()){toast('اكتب ملاحظتك أولاً');return}$('#ftxt').value='';toast('وصلت ملاحظتك للمختص')}
 else if(a==='mq'){S.paused=!S.paused;mq.idle=0;$('#mqb').textContent=S.paused?'▶ تشغيل':'⏸ إيقاف'}
 else if(a==='theme'){setTheme(document.documentElement.dataset.theme==='light'?'dark':'light')}
}

document.addEventListener('click',function(e){var el=e.target.closest('[data-act]');if(el&&!(el.tagName==='A')){act(el)}else if(e.target.id==='ov')closeSheet()});
document.addEventListener('keydown',function(e){if(e.key==='Escape')closeSheet();if(e.key==='Enter'&&e.target.matches('article[data-act]'))act(e.target)});
var saved=null;try{saved=localStorage.getItem('theme')}catch(e){}
setTheme(saved||(matchMedia('(prefers-color-scheme: light)').matches?'light':'dark'));
render();
